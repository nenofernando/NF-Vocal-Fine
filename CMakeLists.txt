cmake_minimum_required(VERSION 3.22)
set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version")
project(NFVocalFine VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_DIR "$ENV{HOME}/JUCE" CACHE PATH "JUCE 9 checkout")
add_subdirectory("${JUCE_DIR}" JUCE)
juce_add_binary_data(FineAssets HEADER_NAME FineAssets.h NAMESPACE FineAssets SOURCES Assets/Logo.png)
# Keep manufacturer/plugin codes and state keys stable across the rename.
juce_add_plugin(NFVocalFine COMPANY_NAME "NF Audio Tools" BUNDLE_ID com.nfaudiotools.vocalnatural PLUGIN_MANUFACTURER_CODE Nfat PLUGIN_CODE Nvnt FORMATS VST3 PRODUCT_NAME "NF Vocal Fine" VST3_CATEGORIES Fx PitchShift COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(NFVocalFine PRIVATE Source/Plugin.cpp)
target_compile_definitions(NFVocalFine PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(NFVocalFine PRIVATE FineAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
option(NF_BUILD_TESTS "Build DSP and VST3 host checks" OFF)
if(NF_BUILD_TESTS)
    juce_add_console_app(VocalFineTests PRODUCT_NAME "VocalFineTests")
    target_sources(VocalFineTests PRIVATE Source/Tests.cpp)
    target_compile_definitions(VocalFineTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
    target_link_libraries(VocalFineTests PRIVATE juce::juce_audio_utils juce::juce_dsp)
    juce_add_console_app(VocalFineEditorChecks PRODUCT_NAME "VocalFineEditorChecks")
    target_sources(VocalFineEditorChecks PRIVATE Source/EditorChecks.cpp)
    target_compile_definitions(VocalFineEditorChecks PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(VocalFineEditorChecks PRIVATE FineAssets juce::juce_audio_utils juce::juce_dsp)
    juce_add_console_app(VocalFineManualChecks PRODUCT_NAME "VocalFineManualChecks")
    target_sources(VocalFineManualChecks PRIVATE Source/ManualChecks.cpp)
    target_compile_definitions(VocalFineManualChecks PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(VocalFineManualChecks PRIVATE FineAssets juce::juce_audio_utils juce::juce_dsp)
endif()
